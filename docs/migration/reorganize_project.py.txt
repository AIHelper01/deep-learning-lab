def write(name, text):
    p = root / name
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(text.rstrip() + '\n', encoding='utf-8')

def rel(target, parent):
    return Path(os.path.relpath(target, parent)).as_posix()

def fix_links(s, old, new):
    def change(m):
        link = m.group(2)
        clean = link.split('#')[0]
        if not clean or re.match(r'\w+://|data:|attachment:',clean): return m.group(0)
        absolute = (root / old).parent.joinpath(clean.replace('\\','/')).resolve()
        if not absolute.is_relative_to(root): return m.group(0)
        key = absolute.relative_to(root).as_posix()
        if key not in mapping: return m.group(0)
        suffix = '#' + link.split('#',1)[1] if '#' in link else ''
        return m.group(1) + rel(root / mapping[key], (root / new).parent) + suffix + m.group(3)
    s = re.sub(r'(\]\()([^\)]+)(\))',change,s)
    def html(m):
        target = (root / old).parent.joinpath(m.group(2)).resolve()
        if target.is_relative_to(root) and target.relative_to(root).as_posix() in mapping:
            return m.group(1)+rel(root/mapping[target.relative_to(root).as_posix()],(root/new).parent)+m.group(3)
        return m.group(0)
    return re.sub(r'(src=["\'])([^"\']+)(["\'])',html,s)
